{{ else if eq .Type "audio" }}
{{ else }} {{- $images := .Resources.ByType "image" -}} {{- $featured := $images.GetMatch "*feature*" -}} {{- $featured = cond (ne $featured nil) $featured ($images.GetMatch "{*cover*,*thumbnail*}") -}} {{- with .Params.featured_image }}{{ $featured = $page.Resources.GetMatch (printf "*%s" .) }}{{ end -}} {{- $title := .Title | plainify -}} {{- with $featured -}} {{- $img := . -}} {{- $thumb := $img -}} {{- $variants := slice -}} {{- $sizes := "" -}} {{- if $fullsize -}} {{- /* Content width: container minus padding, max. 1296 px */ -}} {{- $thumb = partial "image-webp.html" (dict "image" $img "width" 1108) -}} {{- if in (slice "jpeg" "png") $img.MediaType.SubType -}} {{- range slice 640 1108 1600 -}} {{- $variants = $variants | append (partial "image-webp.html" (dict "image" $img "width" .)) -}} {{- end -}} {{- end -}} {{- $sizes = "(min-width: 1400px) 1296px, (min-width: 1200px) 1116px, (min-width: 992px) 936px, (min-width: 768px) 696px, (min-width: 576px) 516px, calc(100vw - 24px)" -}} {{- else -}} {{- /* Start page grid: three columns from md, one column below; 16:9, no upscaling of extra sizes */ -}} {{- $thumb = partial "image-webp.html" (dict "image" $img "fill" "640x360 center") -}} {{- if in (slice "jpeg" "png") $img.MediaType.SubType -}} {{- range slice (slice 400 225) (slice 640 360) (slice 960 540) -}} {{- $w := index . 0 }}{{ $h := index . 1 -}} {{- if or (eq $w 640) (and (ge $img.Width $w) (ge $img.Height $h)) -}} {{- $variants = $variants | append (partial "image-webp.html" (dict "image" $img "fill" (printf "%dx%d center" $w $h))) -}} {{- end -}} {{- end -}} {{- end -}} {{- $sizes = "(min-width: 1400px) 416px, (min-width: 1200px) 356px, (min-width: 992px) 296px, (min-width: 768px) 216px, (min-width: 576px) 516px, calc(100vw - 24px)" -}} {{- end -}} {{- $srcset := slice -}} {{- $seen := slice -}} {{- range $variants -}} {{- if not (in $seen .Width) -}} {{- $srcset = $srcset | append (printf "%s %dw" .RelPermalink .Width) -}} {{- $seen = $seen | append .Width -}} {{- end -}} {{- end }} {{ end }} {{ end }} {{- end -}}
